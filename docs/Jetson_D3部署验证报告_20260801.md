```bash
    hotpot_platform/cloud/supply_chain/manager.py \
    tests/test_d3_integration.py \
    tests/__init__.py

# 2. SCP到Jetson
$ scp d3_integration_deploy.tar.gz root@172.16.1.60:/opt/hotpot-smart-ops/

# 3. 解压 (在Jetson上)
$ cd /opt/hotpot-smart-ops && tar -xzvf d3_integration_deploy.tar.gz

# 4. 备份原始manager.py
$ cp manager.py manager.py.bak.$(date +%Y%m%d%H%M%S)

# 5. 重启Edge UI
$ cd edge/edge-ui && PYTHONPATH=/opt/hotpot-smart-ops nohup python3 main.py &
```
